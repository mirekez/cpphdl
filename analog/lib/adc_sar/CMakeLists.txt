cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(cpphdl_adc_sar LANGUAGES CXX)
  include(CTest)
endif()
get_filename_component(ADC_ANALOG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(cpphdl_adc_sar INTERFACE)
add_library(cpphdl::adc_sar ALIAS cpphdl_adc_sar)
target_include_directories(cpphdl_adc_sar INTERFACE "${ADC_ANALOG_ROOT}" "${ADC_ANALOG_ROOT}/include" "${ADC_ANALOG_ROOT}/../include")
target_compile_features(cpphdl_adc_sar INTERFACE cxx_std_20)
if(BUILD_TESTING)
  find_path(ADC_EIGEN_INCLUDE Eigen/Sparse PATH_SUFFIXES eigen3
    HINTS "${ADC_ANALOG_ROOT}/../build/rx-tooldeps/usr/include/eigen3")
  add_executable(adc_sar_tests tests/test.cpp)
  target_link_libraries(adc_sar_tests PRIVATE cpphdl_adc_sar)
  target_compile_options(adc_sar_tests PRIVATE -Wall -Wextra)
  if(ADC_EIGEN_INCLUDE)
    target_include_directories(adc_sar_tests PRIVATE "${ADC_EIGEN_INCLUDE}")
    target_compile_definitions(adc_sar_tests PRIVATE ADC_USE_EIGEN=1)
  endif()
  set(ADC_SAR_RESULTS "${CMAKE_CURRENT_BINARY_DIR}/results" CACHE PATH "ADC SAR evidence directory")
  foreach(suite IN ITEMS parameters acquisition dac conversion routing comparator leakage timing voltage range convergence sensitivity dynamic settling clock startup mismatch adaptive_input history)
    add_test(NAME analog.lib.adc_sar.${suite} COMMAND adc_sar_tests ${suite} "${ADC_SAR_RESULTS}")
    set_tests_properties(analog.lib.adc_sar.${suite} PROPERTIES LABELS "analog;adc_sar" TIMEOUT 600)
  endforeach()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME analog.lib.adc_sar.ihp_waveform_metrics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/IHPSG13G2/test_compare.py")
  set_tests_properties(analog.lib.adc_sar.ihp_waveform_metrics PROPERTIES LABELS "analog;adc_sar;IHPSG13G2" TIMEOUT 30)

  add_test(NAME analog.lib.adc_sar.ihp_matched_placement COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/IHPSG13G2/test_matched_order.py")
  set_tests_properties(analog.lib.adc_sar.ihp_matched_placement PROPERTIES LABELS "analog;adc_sar;IHPSG13G2" TIMEOUT 30)
  add_test(NAME analog.lib.adc_sar.ihp_loading COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/IHPSG13G2/test_loading.py")
  set_tests_properties(analog.lib.adc_sar.ihp_loading PROPERTIES LABELS "analog;adc_sar;IHPSG13G2" TIMEOUT 30)
  add_executable(adc_sar_ihp_export IHPSG13G2/export.cpp)
  target_link_libraries(adc_sar_ihp_export PRIVATE cpphdl_adc_sar)
  add_test(NAME analog.lib.adc_sar.ihp_model_element_names
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/IHPSG13G2/test_model_element_names.py"
      "$<TARGET_FILE:adc_sar_ihp_export>")
  set_tests_properties(analog.lib.adc_sar.ihp_model_element_names
    PROPERTIES LABELS "analog;adc_sar;IHPSG13G2" TIMEOUT 60)
  find_path(ADC_OSDI_INCLUDE osdi.h HINTS "${ADC_ANALOG_ROOT}/../build/adc-model-closure/osdi-sdk")
  if(ADC_EIGEN_INCLUDE AND ADC_OSDI_INCLUDE)
    add_executable(adc_sar_ihp_native IHPSG13G2/native.cpp)
    target_link_libraries(adc_sar_ihp_native PRIVATE cpphdl_adc_sar ${CMAKE_DL_LIBS})
    target_include_directories(adc_sar_ihp_native PRIVATE "${ADC_EIGEN_INCLUDE}" "${ADC_OSDI_INCLUDE}")
  endif()
  add_test(NAME analog.lib.adc_sar.metrics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_metrics.py")
  set_tests_properties(analog.lib.adc_sar.metrics PROPERTIES LABELS "analog;adc_sar" TIMEOUT 30)
  add_test(NAME analog.lib.adc_sar.qualification COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_qualification.py")
  set_tests_properties(analog.lib.adc_sar.qualification PROPERTIES LABELS "analog;adc_sar" TIMEOUT 30)
  add_test(NAME analog.lib.adc_sar.transition_search COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_transitions.py")
  set_tests_properties(analog.lib.adc_sar.transition_search PROPERTIES LABELS "analog;adc_sar" TIMEOUT 30)
  add_test(NAME analog.lib.adc_sar.transitions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/transitions.py"

  target_compile_options(adc_sar_noise_tests PRIVATE -Wall -Wextra)
  if(ADC_EIGEN_INCLUDE)
    target_include_directories(adc_sar_noise_tests PRIVATE "${ADC_EIGEN_INCLUDE}")
    target_compile_definitions(adc_sar_noise_tests PRIVATE ADC_USE_EIGEN=1)
  endif()
  foreach(suite IN ITEMS model rc)
    add_test(NAME analog.lib.adc_sar.noise_${suite} COMMAND adc_sar_noise_tests ${suite} "${ADC_SAR_RESULTS}/noise")
    set_tests_properties(analog.lib.adc_sar.noise_${suite} PROPERTIES LABELS "analog;adc_sar;noise" TIMEOUT 600)
  endforeach()
  add_test(NAME analog.lib.adc_sar.noise_metrics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_noise_calibration.py")
  set_tests_properties(analog.lib.adc_sar.noise_metrics PROPERTIES LABELS "analog;adc_sar;noise" TIMEOUT 60)
  add_test(NAME analog.lib.adc_sar.noise_adc COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/noise_adc.py"
    --bench "$<TARGET_FILE:adc_sar_tests>" --noise-bench "$<TARGET_FILE:adc_sar_noise_tests>" --work "${ADC_SAR_RESULTS}/noise_adc")
  set_tests_properties(analog.lib.adc_sar.noise_adc PROPERTIES LABELS "analog;adc_sar;noise" TIMEOUT 7200)
  add_executable(adc_sar_edge_tests "${ADC_ANALOG_ROOT}/tests/edge_refinement.cpp")
  target_link_libraries(adc_sar_edge_tests PRIVATE cpphdl_adc_sar)

  foreach(suite IN ITEMS frequency amplitude impedance kickback psrr acquisition evaluation jitter power)
    add_test(NAME analog.lib.adc_sar.characterize_${suite} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/characterization.py"
      --bench "$<TARGET_FILE:adc_sar_tests>" --suite ${suite} --work "${ADC_SAR_RESULTS}/characterization/${suite}")
    set_tests_properties(analog.lib.adc_sar.characterize_${suite} PROPERTIES LABELS "analog;adc_sar;characterization" TIMEOUT 7200)

  add_test(NAME analog.lib.adc_sar.characterization_metrics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_characterization_metrics.py")
  set_tests_properties(analog.lib.adc_sar.characterization_metrics PROPERTIES LABELS "analog;adc_sar;characterization" TIMEOUT 60)
