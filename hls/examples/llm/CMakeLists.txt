add_hls_native(hls_llm_product64 WeightProduct.cpp LLM_DDR_BITS=64)

foreach(example MatrixMath ScalarMath TiledMatVec StreamingMatVec)
    add_hls_native(hls_llm_${example} "${example}.cpp")
    add_test(NAME hls_llm_${example}_native COMMAND "${HLS_NATIVE_BINARY}")
endforeach()
foreach(config depth1 depth13 window2)
    set(depth 8)
    set(window 8)
    if(config STREQUAL "depth1")
        set(depth 1)
    elseif(config STREQUAL "depth13")
        set(depth 13)
    else()
        set(window 2)
    endif()
    add_hls_native(hls_llm_StreamingMatVec_${config} StreamingMatVec.cpp LLM_TILE_DEPTH=${depth} LLM_READ_WINDOW=${window})
    add_test(NAME hls_llm_StreamingMatVec_${config}_native COMMAND "${HLS_NATIVE_BINARY}")
    if(HLS_VERILATOR)
        set(flows rtl)
        if(config STREQUAL "depth1")
            list(APPEND flows gates)
        endif()
        foreach(flow ${flows})
            set(options --top StreamingMatVec --depth ${depth} --window ${window})
            if(flow STREQUAL "gates")
                list(APPEND options --gates)
            endif()
            add_test(NAME hls_llm_StreamingMatVec_${config}_${flow} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Run.py"
                --cpphdl "$<TARGET_FILE:cpphdl>" --cxx "${CMAKE_CXX_COMPILER}"
                --verilator "${HLS_VERILATOR}" --work "${CMAKE_CURRENT_BINARY_DIR}/StreamingMatVec_${config}_${flow}" ${options})
            set_tests_properties(hls_llm_StreamingMatVec_${config}_${flow} PROPERTIES TIMEOUT 2400 LABELS "hls;llm")
        endforeach()
    endif()
endforeach()

        set_tests_properties(hls_llm_${example}_${flow} PROPERTIES TIMEOUT 2400 LABELS "hls;llm")

            --verilator "${HLS_VERILATOR}" --work "${CMAKE_CURRENT_BINARY_DIR}/${flow}" ${options})
        set_tests_properties(hls_llm_${flow} PROPERTIES TIMEOUT 2400 LABELS "hls;llm")
